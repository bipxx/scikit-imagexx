cmake_minimum_required(VERSION 3.20)
project(scikit_imagexx VERSION 0.1.0 LANGUAGES CXX)
option(SKIMAGEXX_BUILD_PYTHON "Build the NumPy bindings" OFF)
option(SKIMAGEXX_OPENMP "Enable CPU parallelism" ON)
option(SKIMAGEXX_AVX2 "Build runtime-dispatched AVX2 kernels on x86" ON)

add_library(skimagexx STATIC src/image.cpp src/convert.cpp src/color.cpp src/perceptual.cpp src/stain.cpp src/distance.cpp src/morphology.cpp src/morphology_boolean.cpp src/threshold.cpp src/local_threshold.cpp src/alpha.cpp src/small_edge.cpp src/farid.cpp src/gaussian_nd.cpp src/integral.cpp src/integrate.cpp src/downscale.cpp src/contrast.cpp src/label_color.cpp src/histogram_matching.cpp src/li.cpp src/adaptive.cpp src/hysteresis.cpp src/sparse.cpp src/median.cpp src/multiotsu.cpp src/clahe.cpp src/gabor.cpp src/gabor_filter.cpp src/symmetric_eigen.cpp src/structure_tensor.cpp)
add_library(skimagexx::skimagexx ALIAS skimagexx)
set_target_properties(skimagexx PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(skimagexx PUBLIC cxx_std_17)
target_include_directories(skimagexx PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>)
if(SKIMAGEXX_OPENMP)
    find_package(OpenMP REQUIRED)
    target_link_libraries(skimagexx PUBLIC OpenMP::OpenMP_CXX)
endif()
if(SKIMAGEXX_AVX2 AND CMAKE_SYSTEM_PROCESSOR MATCHES "AMD64|amd64|x86_64|i.86")
    target_sources(skimagexx PRIVATE src/simd_avx2.cpp src/simd_convert_avx2.cpp src/simd_color_avx2.cpp src/simd_morphology_avx2.cpp src/simd_stain_avx2.cpp src/simd_threshold_avx2.cpp src/simd_local_threshold_avx2.cpp src/simd_alpha_avx2.cpp src/simd_small_edge_avx2.cpp src/simd_farid_avx2.cpp src/simd_gaussian_nd_avx2.cpp src/simd_integral_avx2.cpp src/simd_integrate_avx2.cpp src/simd_downscale_avx2.cpp src/simd_contrast_avx2.cpp src/simd_label_avx2.cpp src/simd_histogram_matching_avx2.cpp src/simd_li_avx2.cpp src/simd_adaptive_avx2.cpp src/simd_hysteresis_avx2.cpp src/simd_sparse_avx2.cpp src/simd_median_avx2.cpp src/simd_multiotsu_avx2.cpp src/simd_clahe_avx2.cpp src/simd_gabor_avx2.cpp src/simd_gabor_filter_avx2.cpp src/simd_symmetric_eigen_avx2.cpp src/simd_structure_tensor_avx2.cpp src/simd_threshold_fma.cpp src/simd_distance_fma.cpp)
    target_compile_definitions(skimagexx PRIVATE SKIMAGEXX_AVX2=1)
    if(MSVC)
        set_source_files_properties(src/simd_avx2.cpp src/simd_convert_avx2.cpp src/simd_color_avx2.cpp src/simd_morphology_avx2.cpp src/simd_stain_avx2.cpp src/simd_threshold_avx2.cpp src/simd_local_threshold_avx2.cpp src/simd_alpha_avx2.cpp src/simd_small_edge_avx2.cpp src/simd_farid_avx2.cpp src/simd_gaussian_nd_avx2.cpp src/simd_integral_avx2.cpp src/simd_integrate_avx2.cpp src/simd_downscale_avx2.cpp src/simd_contrast_avx2.cpp src/simd_label_avx2.cpp src/simd_histogram_matching_avx2.cpp src/simd_li_avx2.cpp src/simd_adaptive_avx2.cpp src/simd_hysteresis_avx2.cpp src/simd_sparse_avx2.cpp src/simd_median_avx2.cpp src/simd_multiotsu_avx2.cpp src/simd_clahe_avx2.cpp src/simd_gabor_avx2.cpp src/simd_gabor_filter_avx2.cpp src/simd_symmetric_eigen_avx2.cpp src/simd_structure_tensor_avx2.cpp src/simd_threshold_fma.cpp src/simd_distance_fma.cpp PROPERTIES COMPILE_OPTIONS "/arch:AVX2;/fp:strict")
    else()
        set_source_files_properties(src/simd_avx2.cpp src/simd_convert_avx2.cpp src/simd_color_avx2.cpp src/simd_morphology_avx2.cpp src/simd_stain_avx2.cpp src/simd_threshold_avx2.cpp src/simd_local_threshold_avx2.cpp src/simd_alpha_avx2.cpp src/simd_small_edge_avx2.cpp src/simd_farid_avx2.cpp src/simd_gaussian_nd_avx2.cpp src/simd_integral_avx2.cpp src/simd_integrate_avx2.cpp src/simd_downscale_avx2.cpp src/simd_contrast_avx2.cpp src/simd_label_avx2.cpp src/simd_histogram_matching_avx2.cpp src/simd_li_avx2.cpp src/simd_adaptive_avx2.cpp src/simd_hysteresis_avx2.cpp src/simd_sparse_avx2.cpp src/simd_median_avx2.cpp src/simd_multiotsu_avx2.cpp src/simd_clahe_avx2.cpp src/simd_gabor_avx2.cpp src/simd_gabor_filter_avx2.cpp src/simd_symmetric_eigen_avx2.cpp src/simd_structure_tensor_avx2.cpp src/simd_threshold_fma.cpp src/simd_distance_fma.cpp PROPERTIES COMPILE_OPTIONS "-mavx2;-ffp-contract=off")
        set_source_files_properties(src/simd_threshold_fma.cpp src/simd_distance_fma.cpp PROPERTIES COMPILE_OPTIONS "-mavx2;-mfma;-ffp-contract=off")
    endif()
endif()
if(MSVC)
    set_source_files_properties(src/gaussian_nd.cpp src/integral.cpp src/integrate.cpp src/downscale.cpp src/contrast.cpp src/label_color.cpp src/histogram_matching.cpp src/li.cpp src/adaptive.cpp src/hysteresis.cpp src/sparse.cpp src/median.cpp src/multiotsu.cpp src/clahe.cpp src/gabor.cpp src/gabor_filter.cpp src/symmetric_eigen.cpp src/structure_tensor.cpp PROPERTIES COMPILE_OPTIONS "/fp:strict")
    target_compile_options(skimagexx PRIVATE /W4 /fp:precise)
else()
    set_source_files_properties(src/gaussian_nd.cpp src/integral.cpp src/integrate.cpp src/downscale.cpp src/contrast.cpp src/label_color.cpp src/histogram_matching.cpp src/li.cpp src/adaptive.cpp src/hysteresis.cpp src/sparse.cpp src/median.cpp src/multiotsu.cpp src/clahe.cpp src/gabor.cpp src/gabor_filter.cpp src/symmetric_eigen.cpp src/structure_tensor.cpp PROPERTIES COMPILE_OPTIONS "-ffp-contract=off")
    target_compile_options(skimagexx PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(SKIMAGEXX_BUILD_PYTHON)
    find_package(Python COMPONENTS Interpreter Development.Module NumPy REQUIRED)
    find_package(pybind11 CONFIG REQUIRED)
    pybind11_add_module(_core src/bindings.cpp src/bind_gabor.cpp src/bind_extra.cpp src/bind_symmetric_eigen.cpp src/bind_structure_tensor.cpp)
    target_link_libraries(_core PRIVATE skimagexx Python::NumPy)
    install(TARGETS _core LIBRARY DESTINATION skimagexx RUNTIME DESTINATION skimagexx)
endif()

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/skimagexxConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/skimagexxConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/skimagexx)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/skimagexxConfigVersion.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/skimagexxConfig.cmake
              ${CMAKE_CURRENT_BINARY_DIR}/skimagexxConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/skimagexx)
install(FILES LICENSE third_party/scikit-image-LICENSE.txt third_party/svml-LICENSE.txt third_party/numpy-LICENSE.txt third_party/scipy-ndimage-LICENSE.txt third_party/scipy-rank1d-LICENSE.txt third_party/lapack-LICENSE.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/scikit-imagexx)
install(TARGETS skimagexx EXPORT skimagexxTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT skimagexxTargets NAMESPACE skimagexx:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/skimagexx)
