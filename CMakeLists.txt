cmake_minimum_required(VERSION 3.20)
project(scikit_imagexx VERSION 0.1.0 LANGUAGES CXX)
option(SKIMAGEXX_BUILD_PYTHON "Build the NumPy bindings" OFF)
option(SKIMAGEXX_OPENMP "Enable CPU parallelism" ON)
option(SKIMAGEXX_AVX2 "Build runtime-dispatched AVX2 kernels on x86" ON)

add_library(skimagexx STATIC src/image.cpp src/convert.cpp src/color.cpp src/perceptual.cpp src/stain.cpp src/distance.cpp src/morphology.cpp src/morphology_boolean.cpp src/threshold.cpp src/local_threshold.cpp src/alpha.cpp src/small_edge.cpp src/farid.cpp src/gaussian_nd.cpp src/integral.cpp src/integrate.cpp src/downscale.cpp)
add_library(skimagexx::skimagexx ALIAS skimagexx)
set_target_properties(skimagexx PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(skimagexx PUBLIC cxx_std_17)
target_include_directories(skimagexx PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>)
if(SKIMAGEXX_OPENMP)
    find_package(OpenMP REQUIRED)
    target_link_libraries(skimagexx PUBLIC OpenMP::OpenMP_CXX)
endif()
if(SKIMAGEXX_AVX2 AND CMAKE_SYSTEM_PROCESSOR MATCHES "AMD64|amd64|x86_64|i.86")
    target_sources(skimagexx PRIVATE src/simd_avx2.cpp src/simd_convert_avx2.cpp src/simd_color_avx2.cpp src/simd_morphology_avx2.cpp src/simd_stain_avx2.cpp src/simd_threshold_avx2.cpp src/simd_local_threshold_avx2.cpp src/simd_alpha_avx2.cpp src/simd_small_edge_avx2.cpp src/simd_farid_avx2.cpp src/simd_gaussian_nd_avx2.cpp src/simd_integral_avx2.cpp src/simd_integrate_avx2.cpp src/simd_downscale_avx2.cpp src/simd_threshold_fma.cpp)
    target_compile_definitions(skimagexx PRIVATE SKIMAGEXX_AVX2=1)
    if(MSVC)
        set_source_files_properties(src/simd_avx2.cpp src/simd_convert_avx2.cpp src/simd_color_avx2.cpp src/simd_morphology_avx2.cpp src/simd_stain_avx2.cpp src/simd_threshold_avx2.cpp src/simd_local_threshold_avx2.cpp src/simd_alpha_avx2.cpp src/simd_small_edge_avx2.cpp src/simd_farid_avx2.cpp src/simd_gaussian_nd_avx2.cpp src/simd_integral_avx2.cpp src/simd_integrate_avx2.cpp src/simd_downscale_avx2.cpp src/simd_threshold_fma.cpp PROPERTIES COMPILE_OPTIONS "/arch:AVX2;/fp:strict")
    else()
        set_source_files_properties(src/simd_avx2.cpp src/simd_convert_avx2.cpp src/simd_color_avx2.cpp src/simd_morphology_avx2.cpp src/simd_stain_avx2.cpp src/simd_threshold_avx2.cpp src/simd_local_threshold_avx2.cpp src/simd_alpha_avx2.cpp src/simd_small_edge_avx2.cpp src/simd_farid_avx2.cpp src/simd_gaussian_nd_avx2.cpp src/simd_integral_avx2.cpp src/simd_integrate_avx2.cpp src/simd_downscale_avx2.cpp src/simd_threshold_fma.cpp PROPERTIES COMPILE_OPTIONS "-mavx2;-ffp-contract=off")
        set_source_files_properties(src/simd_threshold_fma.cpp PROPERTIES COMPILE_OPTIONS "-mavx2;-mfma;-ffp-contract=off")
    endif()
endif()
if(MSVC)
    set_source_files_properties(src/gaussian_nd.cpp src/integral.cpp src/integrate.cpp src/downscale.cpp PROPERTIES COMPILE_OPTIONS "/fp:strict")
    target_compile_options(skimagexx PRIVATE /W4 /fp:precise)
else()
    set_source_files_properties(src/gaussian_nd.cpp src/integral.cpp src/integrate.cpp src/downscale.cpp PROPERTIES COMPILE_OPTIONS "-ffp-contract=off")
    target_compile_options(skimagexx PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(SKIMAGEXX_BUILD_PYTHON)
    find_package(Python COMPONENTS Interpreter Development.Module NumPy REQUIRED)
    find_package(pybind11 CONFIG REQUIRED)
    pybind11_add_module(_core src/bindings.cpp)
    target_link_libraries(_core PRIVATE skimagexx Python::NumPy)
    install(TARGETS _core LIBRARY DESTINATION skimagexx RUNTIME DESTINATION skimagexx)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(test_native tests/test_native.cpp)
    target_link_libraries(test_native PRIVATE skimagexx)
    add_test(NAME native COMMAND test_native)
endif()

include(GNUInstallDirs)
install(FILES LICENSE third_party/scikit-image-LICENSE.txt third_party/numpy-LICENSE.txt third_party/scipy-ndimage-LICENSE.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/scikit-imagexx)
install(TARGETS skimagexx EXPORT skimagexxTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT skimagexxTargets NAMESPACE skimagexx:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/skimagexx)
