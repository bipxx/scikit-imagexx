cmake_minimum_required(VERSION 3.20)
project(skimagexx_consumer LANGUAGES CXX)
set(SKIMAGEXX_FIND_VERSION "0.1" CACHE STRING "Requested package version")
find_package(skimagexx ${SKIMAGEXX_FIND_VERSION} CONFIG REQUIRED)

cmake_path(ABSOLUTE_PATH SKIMAGEXX_EXPECT_PREFIX NORMALIZE OUTPUT_VARIABLE _prefix)
cmake_path(IS_PREFIX _prefix "${skimagexx_DIR}" NORMALIZE _expected_config)
if(NOT _expected_config)
    message(FATAL_ERROR "Package config escaped the relocated prefix: ${skimagexx_DIR}")
endif()
string(TOUPPER "${CMAKE_BUILD_TYPE}" _config)
if(NOT _config)
    set(_config RELEASE)
endif()
get_target_property(_library skimagexx::skimagexx IMPORTED_LOCATION_${_config})
cmake_path(IS_PREFIX _prefix "${_library}" NORMALIZE _expected_library)
if(NOT _expected_library)
    message(FATAL_ERROR "Library escaped the relocated prefix: ${_library}")
endif()
get_target_property(_includes skimagexx::skimagexx INTERFACE_INCLUDE_DIRECTORIES)
foreach(_include IN LISTS _includes)
    cmake_path(IS_PREFIX _prefix "${_include}" NORMALIZE _expected_include)
    if(NOT _expected_include)
        message(FATAL_ERROR "Header path escaped the relocated prefix: ${_include}")
    endif()
endforeach()

get_target_property(_links skimagexx::skimagexx INTERFACE_LINK_LIBRARIES)
if(SKIMAGEXX_EXPECT_OPENMP)
    if(NOT "OpenMP::OpenMP_CXX" IN_LIST _links)
        message(FATAL_ERROR "OpenMP dependency is missing")
    endif()
elseif("OpenMP::OpenMP_CXX" IN_LIST _links)
    message(FATAL_ERROR "Portable package unexpectedly requires OpenMP")
endif()

# Exercise the installed public headers and static library for the newer core
# families as an independent client, without Python or source-tree include paths.
find_package(Threads REQUIRED)
foreach(_family IN ITEMS components component_consumers consumer_layout metrics footprint_dense rank rank_distribution rank_intensity rank_alias rank_statistics rank_percentile rank_bilateral rank_decisions projective tv_chambolle tv_reductions tv_denoise tv_bregman max_tree matrix_transform reconstruction)
    add_executable(${_family}_consumer ../test_${_family}_native.cpp)
    target_link_libraries(${_family}_consumer PRIVATE skimagexx::skimagexx Threads::Threads)
    set(_family_test_args)
    if(_family STREQUAL "max_tree")
        include("${CMAKE_CURRENT_SOURCE_DIR}/max_tree_source_oracle.cmake")
        list(APPEND _family_test_args "${SKIMAGEXX_MAX_TREE_SOURCE_ORACLE}")
    endif()
    if(_family STREQUAL "reconstruction")
        include("${CMAKE_CURRENT_SOURCE_DIR}/reconstruction_source_oracle.cmake")
        list(APPEND _family_test_args "${SKIMAGEXX_RECONSTRUCTION_SOURCE_ORACLE}")
    endif()
    add_test(NAME installed_${_family}_consumer COMMAND ${_family}_consumer ${_family_test_args})
endforeach()

# H203 uses both unchanged source-derived integer masks and exact real controls.
add_executable(h_extrema_consumer h_extrema_consumer.cpp)
target_link_libraries(h_extrema_consumer PRIVATE skimagexx::skimagexx Threads::Threads)
if(MSVC)
    target_compile_options(h_extrema_consumer PRIVATE /UNDEBUG /fp:strict)
else()
    target_compile_options(h_extrema_consumer PRIVATE -UNDEBUG -ffp-contract=off -fno-fast-math -frounding-math)
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/h_extrema_source_oracle.cmake")
add_test(NAME installed_h_extrema_consumer COMMAND h_extrema_consumer "${SKIMAGEXX_H_EXTREMA_SOURCE_ORACLE}")
