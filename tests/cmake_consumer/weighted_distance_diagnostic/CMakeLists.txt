cmake_minimum_required(VERSION 3.20)
project(weighted_distance_installed_consumer LANGUAGES CXX)
find_package(skimagexx 0.1 CONFIG REQUIRED)
if(NOT SKIMAGEXX_EXPECT_PREFIX)
    message(FATAL_ERROR "Explicit installed prefix required")
endif()
cmake_path(ABSOLUTE_PATH SKIMAGEXX_EXPECT_PREFIX NORMALIZE OUTPUT_VARIABLE _prefix)
cmake_path(IS_PREFIX _prefix "${skimagexx_DIR}" NORMALIZE _config_inside)
get_target_property(_library skimagexx::skimagexx IMPORTED_LOCATION_RELEASE)
get_target_property(_includes skimagexx::skimagexx INTERFACE_INCLUDE_DIRECTORIES)
cmake_path(IS_PREFIX _prefix "${_library}" NORMALIZE _library_inside)
if(NOT _config_inside OR NOT _library_inside)
    message(FATAL_ERROR "Imported SDK config/library escaped expected installed prefix")
endif()
foreach(_include IN LISTS _includes)
    cmake_path(IS_PREFIX _prefix "${_include}" NORMALIZE _header_inside)
    if(NOT _header_inside)
        message(FATAL_ERROR "Imported SDK headers escaped expected installed prefix")
    endif()
endforeach()
message(STATUS "Installed weighted SDK library=${_library}; headers=${_includes}")
add_executable(weighted_distance_consumer weighted_distance_consumer.cpp)
target_compile_features(weighted_distance_consumer PRIVATE cxx_std_17)
target_link_libraries(weighted_distance_consumer PRIVATE skimagexx::skimagexx)
if(MSVC)
    target_compile_options(weighted_distance_consumer PRIVATE /UNDEBUG /fp:strict)
else()
    target_compile_options(weighted_distance_consumer PRIVATE -UNDEBUG -ffp-contract=off -fno-fast-math)
    target_link_options(weighted_distance_consumer PRIVATE "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/weighted-distance.map")
endif()
enable_testing()
add_test(NAME installed_weighted_distance COMMAND weighted_distance_consumer)
add_executable(weighted_distance_load_paths ../../test_distance_load_paths.cpp)
target_compile_features(weighted_distance_load_paths PRIVATE cxx_std_17)
target_link_libraries(weighted_distance_load_paths PRIVATE skimagexx::skimagexx)
if(MSVC)
    target_compile_options(weighted_distance_load_paths PRIVATE /fp:strict)
else()
    target_compile_options(weighted_distance_load_paths PRIVATE -ffp-contract=off)
endif()
add_test(NAME installed_distance_load_paths COMMAND weighted_distance_load_paths)
