cmake_minimum_required(VERSION 3.18)
project(inverse_gaussian_gradient_complete_consumer LANGUAGES CXX)
find_package(skimagexx CONFIG REQUIRED)
add_executable(inverse_gaussian_gradient_complete_consumer ../../test_inverse_gaussian_gradient_sdk.cpp)
target_link_libraries(inverse_gaussian_gradient_complete_consumer PRIVATE skimagexx::skimagexx)
target_compile_features(inverse_gaussian_gradient_complete_consumer PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(inverse_gaussian_gradient_complete_consumer PRIVATE /fp:strict /UNDEBUG)
else()
  target_compile_options(inverse_gaussian_gradient_complete_consumer PRIVATE -ffp-contract=off -fno-fast-math -frounding-math -UNDEBUG)
endif()
