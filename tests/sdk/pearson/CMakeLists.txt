cmake_minimum_required(VERSION 3.18)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(PearsonClient LANGUAGES CXX)
    find_package(skimagexx CONFIG REQUIRED)
    enable_testing()
endif()

add_executable(pearson_client ../../cmake_consumer/pearson_consumer.cpp)
target_compile_features(pearson_client PRIVATE cxx_std_17)
target_link_libraries(pearson_client PRIVATE skimagexx::skimagexx)
if(MSVC)
    target_compile_options(pearson_client PRIVATE /fp:strict)
else()
    target_compile_options(pearson_client PRIVATE -ffp-contract=off -fno-fast-math -frounding-math)
endif()
add_test(NAME installed_pearson_consumer COMMAND pearson_client)
