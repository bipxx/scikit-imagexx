cmake_minimum_required(VERSION 3.18)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(DataCheckerboardClient LANGUAGES CXX)
    find_package(skimagexx CONFIG REQUIRED)
    enable_testing()
endif()

add_executable(data_checkerboard_client ../../cmake_consumer/data_checkerboard_consumer.cpp)
target_compile_features(data_checkerboard_client PRIVATE cxx_std_17)
target_link_libraries(data_checkerboard_client PRIVATE skimagexx::skimagexx)
add_test(NAME installed_data_checkerboard_consumer COMMAND data_checkerboard_client)
