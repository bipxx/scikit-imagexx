cmake_minimum_required(VERSION 3.18)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(CornerOrientationsClient LANGUAGES CXX)
    find_package(skimagexx CONFIG REQUIRED)
    enable_testing()
endif()

add_executable(corner_orientations_client ../../cmake_consumer/corner_orientations_consumer.cpp)
target_compile_features(corner_orientations_client PRIVATE cxx_std_17)
target_link_libraries(corner_orientations_client PRIVATE skimagexx::skimagexx)
if(MSVC)
    target_compile_options(corner_orientations_client PRIVATE /fp:strict)
else()
    target_compile_options(corner_orientations_client PRIVATE -ffp-contract=off -fno-fast-math -frounding-math)
endif()
add_test(NAME installed_corner_orientations_consumer COMMAND corner_orientations_client)
